\sectionlabel{One integrated workbench, three connected products}

\begin{tabularx}{\textwidth}{@{}Y@{\hspace{0.10in}}Y@{\hspace{0.10in}}Y@{}}
  \productcard{1. Static Admissibility}
    {Can this study be trusted enough to pursue?}
    {Screens feature, class, prior, coverage, provenance, and leakage risks before expensive modeling or data search.}
    {A decision card and a concrete route: promote, revise, or reject.}
  &
  \productcard{2. Evidence Ladder}
    {What is the simplest method that is sufficient?}
    {Compares classifiers and filters through the same evidence and posterior contract, adding complexity only for a diagnosed failure mode.}
    {Comparable method scorecards, posterior histories, and a promotion decision.}
  &
  \productcard{3. Corpus Explorer}
    {Which valid hard cases expose the next limitation?}
    {Generates or selects difficult, decision-useful data while checking validity, coverage, balance, and leakage. Search and RL are tools, not unchecked answers.}
    {Selected corpus and stress packets that feed targeted cases back into the ladder.}
\end{tabularx}

\vspace{0.36em}
\noindent{\color{ink}\small\bfseries The operating loop: }\small
screen the study \(\rightarrow\) build and compare evidence \(\rightarrow\) find valid boundary cases \(\rightarrow\) improve the study.\par
The products share one study definition, evidence/posterior contract, and decision language; they are not three disconnected tools.

\begin{minipage}[t]{0.485\textwidth}
  \sectionlabel{What a sponsor receives}
  \small
  \valuebullet{\textbf{Portable study definitions} that make classes, features, priors, data sources, and intended claims explicit.}
  \valuebullet{\textbf{Decision packets} with provenance, coverage, leakage, prior, and failure diagnostics rather than a black-box score.}
  \valuebullet{\textbf{Comparable evidence} across interpretable, physics-aware, benchmark, and learned method families.}
  \valuebullet{\textbf{Purpose-built hard cases} and corpus manifests that turn an observed weakness into the next controlled test.}
\end{minipage}%
\hfill
\begin{minipage}[t]{0.485\textwidth}
  \sectionlabel{Why it creates value}
  \small
  \valuebullet{\textbf{Reduces avoidable spend:} catches inseparable class definitions, weak or leaky features, and fragile priors before compute and model effort scale.}
  \valuebullet{\textbf{Keeps complexity honest:} supports advanced methods when a named failure case justifies them, not because they are fashionable.}
  \valuebullet{\textbf{Speeds learning:} turns a weak result into a specific repair, data-search objective, or method decision.}
  \valuebullet{\textbf{Creates a reusable path to 3D:} preserves study, evidence, artifact, and decision contracts while vector data, features, and dynamics mature.}
\end{minipage}
